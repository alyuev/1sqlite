//  ВАЖНО (1С 7.7): НЕ использовать Возврат внутри Попытка — результат

Перем ТЗ, ЭтоАвтомат, глПутьЛога, глЗагружено;

Функция ПутьЛога()
	Если ПустаяСтрока(глПутьЛога)=1 Тогда
		глПутьЛога = КаталогИБ() + "test_log.txt";
	КонецЕсли;
	Возврат глПутьЛога;
КонецФункции

Процедура НачатьЛог()
	Попытка
		ФСО = СоздатьОбъект("Scripting.FileSystemObject");
		П = ФСО.CreateTextFile(ПутьЛога(), 1);
		П.Close();
	Исключение
	КонецПопытки;
КонецПроцедуры

Процедура ВЛог(Текст)
	Попытка
		ФСО = СоздатьОбъект("Scripting.FileSystemObject");
		П = ФСО.OpenTextFile(ПутьЛога(), 8, 1);
		П.WriteLine(Текст);
		П.Close();
	Исключение
	КонецПопытки;
	Сообщить(Текст);
КонецПроцедуры

Функция глОткрытьБазу()
	Если глЗагружено <> 1 Тогда
		ЗагрузитьВнешнююКомпоненту(КаталогИБ() + "1sqlite.dll");
		глЗагружено = 1;
	КонецЕсли;
	База = СоздатьОбъект("SQLiteBase");
	База.Открыть(":memory:");
	Возврат База;
КонецФункции

Функция ВыполнитьЗапросSQLite(Текст, Ист=0, П3=0, Режим=0, П5="")
	Перем База, Зап, сч, Имя, Зн, Результат;
	База = глОткрытьБазу();
	Зап = База.НовыйЗапрос();
	Если ТипЗначенияСтр(Ист) = "СписокЗначений" Тогда
		Для сч = 1 По Ист.РазмерСписка() Цикл
			Имя = "";
			Зн = Ист.ПолучитьЗначение(сч, Имя);
			База.УложитьТЗ(Зн, Имя, 0);
		КонецЦикла;
	КонецЕсли;
	Если П5 = "" Тогда
		Результат = СоздатьОбъект("ТаблицаЗначений");
		Зап.ВыполнитьЗапрос(Текст, Результат);
		Возврат Результат;
	Иначе
		Возврат Зап.ВыполнитьЗапрос(Текст, 0);
	КонецЕсли;
КонецФункции

Процедура Ошибка(ИмяПроц, т="")
	Если ТипЗначенияСтр(ТЗ) = "ТаблицаЗначений" Тогда
		ТЗ.НоваяСтрока();
		ТЗ.Имя_Теста = ИмяПроц;
		ТЗ.Ошибка   = т;
	КонецЕсли;
	ВЛог("   " + ИмяПроц + " => " + т);
КонецПроцедуры
